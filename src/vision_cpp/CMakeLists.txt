cmake_minimum_required(VERSION 3.16)
project(vision_cpp)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
option(BUILD_VIDEO_VIEWER "Build the desktop video window" ON)
set(OPENCV_COMPONENTS core imgproc imgcodecs objdetect wechat_qrcode)
if(BUILD_VIDEO_VIEWER)
  list(APPEND OPENCV_COMPONENTS highgui)
endif()
find_package(OpenCV REQUIRED COMPONENTS ${OPENCV_COMPONENTS})
find_package(Threads REQUIRED)
add_library(camera_core src/camera.cpp src/qr.cpp)
target_include_directories(camera_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(camera_core PUBLIC ${OpenCV_LIBRARIES} Threads::Threads)
add_executable(csi_jpeg_stream src/stream.cpp)
target_link_libraries(csi_jpeg_stream camera_core)
add_executable(rpi_camera_node src/camera_node.cpp)
target_link_libraries(rpi_camera_node camera_core)
ament_target_dependencies(rpi_camera_node rclcpp sensor_msgs std_msgs)
add_executable(qr_ssh_bridge src/ssh_bridge.cpp)
target_link_libraries(qr_ssh_bridge camera_core)
ament_target_dependencies(qr_ssh_bridge rclcpp sensor_msgs)
add_executable(qr_reader_node src/qr_reader.cpp)
target_link_libraries(qr_reader_node camera_core)
ament_target_dependencies(qr_reader_node rclcpp sensor_msgs std_msgs)
add_executable(jpeg_topic_stream src/jpeg_topic_stream.cpp)
target_link_libraries(jpeg_topic_stream camera_core)
ament_target_dependencies(jpeg_topic_stream rclcpp sensor_msgs std_msgs)
if(BUILD_VIDEO_VIEWER)
  add_executable(video_viewer src/video_viewer.cpp)
  target_link_libraries(video_viewer camera_core)
  ament_target_dependencies(video_viewer rclcpp sensor_msgs std_msgs)
  install(TARGETS video_viewer DESTINATION lib/${PROJECT_NAME})
endif()
install(TARGETS csi_jpeg_stream rpi_camera_node qr_ssh_bridge qr_reader_node jpeg_topic_stream
  DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_camera test/test_camera.cpp)
  target_link_libraries(test_camera camera_core)
  ament_add_pytest_test(test_native_nodes test/test_native_nodes.py TIMEOUT 45
    ENV "READER_EXECUTABLE=$<TARGET_FILE:qr_reader_node>"
      "CAMERA_EXECUTABLE=$<TARGET_FILE:rpi_camera_node>" "ROS_DOMAIN_ID=178"
      "RELAY_EXECUTABLE=$<TARGET_FILE:jpeg_topic_stream>"
      "ROS_LOCALHOST_ONLY=1")
endif()
ament_package()
